# Fix USB permissions for the Android phone (Pixel, vendor 18d1)
# Run these three commands, then unplug and replug the phone.

echo 'SUBSYSTEM=="usb", ATTR{idVendor}=="18d1", MODE="0666", GROUP="plugdev"' | sudo tee /etc/udev/rules.d/51-android.rules
sudo udevadm control --reload-rules
sudo udevadm trigger

# Afterwards: unplug + replug the phone, accept the
# "Allow USB debugging" popup on the phone if it appears.
